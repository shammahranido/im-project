<?php
include_once 'admin-db.php';
include_once 'all-process.php';

$statusOptions = ['In Progress', 'Cancelled', 'Completed']; // Define status options

try {
    $database = new Database('localhost', 'dvmf', 'postgres', 'admin123');
    $pdo = $database->getPDO();

    $processAppoint = new AllProcess($pdo);

    if (isset($_GET['appoint_id'])) {
        $appointId = $_GET['appoint_id'];

        // Fetch appointment details before deleting
        $query = 'SELECT appointment.customer_id, appointment.appoint_date, appointment.appoint_status,
                         customer.customer_name, customer.cus_num_dog, customer.cus_num_cat 
                  FROM appointment 
                  JOIN customer ON appointment.customer_id = customer.customer_id 
                  WHERE appointment.appoint_id = :appointId';

        $statement = $pdo->prepare($query);
        $statement->bindParam(':appointId', $appointId, PDO::PARAM_INT);
        $statement->execute();
        $appointmentDetails = $statement->fetch(PDO::FETCH_ASSOC);

   
      
        if ($appointmentDetails['appoint_status'] == 'Cancelled' || $appointmentDetails['appoint_status'] == 'In Progress'){
            $deleted = $processAppoint->deleteAppointment($appointId);
            header('Location: admin-menu.php?service=list-rabies-appointment');
            exit();
        }
        // Insert into VACCINATION table
        if (!empty($appointmentDetails) && $appointmentDetails['appoint_status'] == 'Completed') {
            $insertQuery = 'INSERT INTO VACCINATION (customer_id, vac_date) 
                            VALUES (:customerId, :appointDate)';
            
            $insertStatement = $pdo->prepare($insertQuery);
            $insertStatement->bindParam(':customerId', $appointmentDetails['customer_id'], PDO::PARAM_INT);
            $insertStatement->bindParam(':appointDate', $appointmentDetails['appoint_date'], PDO::PARAM_STR);
            $inserted = $insertStatement->execute();

            // Insert into HISTORY table
            if ($inserted) {
                $vacId = $pdo->lastInsertId();
                $insertHistoryQuery = 'INSERT INTO history (customer_id, vac_id, his_cus_name, his_num_cat, his_num_dog) 
                                       VALUES (:customerId, :vacId, :cusName, :cusNumCat, :cusNumDog)';
                
                $insertHistoryStatement = $pdo->prepare($insertHistoryQuery);
                $insertHistoryStatement->bindParam(':vacId', $vacId, PDO::PARAM_INT);
                $insertHistoryStatement->bindParam(':customerId', $appointmentDetails['customer_id'], PDO::PARAM_INT);
                $insertHistoryStatement->bindParam(':cusName', $appointmentDetails['customer_name'], PDO::PARAM_INT);
                $insertHistoryStatement->bindParam(':cusNumCat', $appointmentDetails['cus_num_cat'], PDO::PARAM_INT);
                $insertHistoryStatement->bindParam(':cusNumDog', $appointmentDetails['cus_num_dog'], PDO::PARAM_INT);
                $historyInserted = $insertHistoryStatement->execute();

                if ($historyInserted) {
                    // Redirect upon successful deletion, vaccination, and history insertion
                    header('Location: admin-menu.php?service=list-rabies-appointment');
                    $deleted = $processAppoint->deleteAppointment($appointId);
                    exit();
                } else {
                    echo "Failed to insert into history table";
                    exit();
                }
            } else {
                echo "Failed to insert into VACCINATION table";
                exit();
            }
        } else {
            echo "Appointment deletion failed or appointment details not found";
            exit();
        }
    }

    // Initialize variables for listing appointments
    $searchName = isset($_GET['search_name']) ? htmlspecialchars($_GET['search_name']) : '';
    $searchStatus = isset($_GET['search_status']) ? $_GET['search_status'] : '';
    $appointments = [];

    // Construct query based on search criteria
    $query = 'SELECT appointment.*, customer.customer_name, customer.cus_num_dog, customer.cus_num_cat 
              FROM appointment 
              JOIN customer ON appointment.customer_id = customer.customer_id 
              WHERE appointment.service_id = :id';

    $params = ['id' => 1];

    if (!empty($searchName)) {
        $query .= ' AND customer.customer_name ILIKE :searchName';
        $params['searchName'] = '%' . $searchName . '%';
    }

    if (!empty($searchStatus)) {
        $query .= ' AND appointment.appoint_status = :status';
        $params['status'] = $searchStatus;
    }

    $query .= ' ORDER BY appointment.appoint_date ASC';

    $statement = $pdo->prepare($query);
    $statement->execute($params);
    $appointments = $statement->fetchAll(PDO::FETCH_ASSOC);
} catch (PDOException $e) {
    echo "Query failed: " . $e->getMessage();
    die();
}
